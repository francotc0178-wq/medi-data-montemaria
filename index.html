async function enviarFormulario() { const btn = document.querySelector('button[type="button"]') || document.querySelector('button'); // Captura todos los campos de tu formulario original const email = document.getElementById('email')?.value.trim(); const nombre = document.getElementById('nombre')?.value.trim(); const rut = document.getElementById('rut')?.value.trim(); const observaciones = document.getElementById('observaciones')?.value.trim(); if (!email) { alert("Por favor ingresa un correo electrónico."); return; } if (btn) { btn.disabled = true; btn.innerText = "Enviando..."; } try { const response = await fetch('https://medi-data-montemaria.onrender.com/api/enviar-pdf', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ email: email, nombre: nombre || 'Paciente', rut: rut || 'Sin RUT', datos: observaciones || 'Sin observaciones' }) }); const result = await response.json(); if (response.ok && result.success) { alert("¡Éxito! La ficha médica ha sido enviada a tu correo."); } else { alert("Error: " + (result.error || "No se pudo enviar el correo.")); } } catch (err) { alert("Error de conexión: " + err.message); } finally { if (btn) { btn.disabled = false; btn.innerText = "Enviar Ficha Médica al Correo"; } } }
