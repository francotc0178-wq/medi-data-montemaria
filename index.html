async function generarYEnviarPDF() { const email = document.getElementById('email')?.value.trim(); const nombre = document.getElementById('nombre')?.value.trim() || 'Paciente'; const rut = document.getElementById('rut')?.value.trim() || 'Sin RUT'; if (!email) { alert("Por favor ingresa un correo electrónico."); return; } try { const response = await fetch('https://medi-data-montemaria.onrender.com/api/enviar-pdf', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ email: email, nombre: nombre, rut: rut, datos: "Ficha médica procesada desde el sistema web." }) }); const data = await response.json(); if (response.ok && data.success) { alert("¡Ficha médica enviada con éxito a tu correo!"); } else { alert("Error del servidor: " + (data.error || "No se pudo enviar.")); } } catch (err) { alert("Error de conexión: " + err.message); } }
